/******************************************************************************
 * SECTION 11: JAVASCRIPT - CONTEXT DETECTION
 ******************************************************************************/

function detectProblemContext(message) {
    const lowerMsg = message.toLowerCase();
    const contexts = [];
    
    if (lowerMsg.match(/\b(i feel|i'm|i am|myself|alone|lonely|identity|purpose|meaning|confidence|insecure|not good enough|self|personal|life)\b/)) {
        contexts.push('personal');
    }
    
    if (lowerMsg.match(/\b(work|job|career|boss|manager|colleague|coworker|office|deadline|project|meeting|promotion|salary|quit|fired|workload|overwhelm.*work)\b/)) {
        contexts.push('work');
    }
    
    if (lowerMsg.match(/\b(partner|boyfriend|girlfriend|spouse|husband|wife|relationship|dating|marriage|break.*up|fight|argument|trust|cheat|listen)\b/)) {
        contexts.push('relationship');
    }
    
    if (lowerMsg.match(/\b(health|sleep|insomnia|anxiety|anxious|stress|depressed|depression|tired|exhausted|mental|therapy|panic|worry)\b/)) {
        contexts.push('health');
    }
    
    const uniqueContexts = [...new Set(contexts)];
    
    if (uniqueContexts.length === 0) {
        uniqueContexts.push('personal');
    }
    
    return uniqueContexts.slice(0, 2);
}

/******************************************************************************
 * SECTION 12: JAVASCRIPT - EMOTION ANALYSIS
 ******************************************************************************/

function generateMockAnalysis(message) {
    const lowerMsg = message.toLowerCase();
    
    let emotion = 'neutral';
    if (lowerMsg.includes('never') || lowerMsg.includes('always') || lowerMsg.includes('hate')) {
        emotion = 'frustrated';
    } else if (lowerMsg.includes('sad') || lowerMsg.includes('hurt') || lowerMsg.includes('down')) {
        emotion = 'sad';
    } else if (lowerMsg.includes('love') || lowerMsg.includes('happy') || lowerMsg.includes('excited')) {
        emotion = 'happy';
    } else if (lowerMsg.includes('worried') || lowerMsg.includes('anxious') || lowerMsg.includes('nervous')) {
        emotion = 'anxious';
    } else if (lowerMsg.includes('angry') || lowerMsg.includes('mad') || lowerMsg.includes('furious')) {
        emotion = 'angry';
    } else if (lowerMsg.includes('stressed') || lowerMsg.includes('overwhelm')) {
        emotion = 'stressed';
    }
    
    let tone = 'reflective';
    if (lowerMsg.includes('you never') || lowerMsg.includes('you always')) {
        tone = 'accusatory';
    } else if (lowerMsg.includes('fine') || lowerMsg.includes('whatever')) {
        tone = 'passive-aggressive';
    } else if (emotion === 'happy' || lowerMsg.includes('thank') || lowerMsg.includes('appreciate')) {
        tone = 'warm';
    } else if (lowerMsg.includes('need help') || lowerMsg.includes('struggling')) {
        tone = 'vulnerable';
    }
    
    const concerns = [];
    if (lowerMsg.includes('never') || lowerMsg.includes('always')) concerns.push('absolute language');
    if (lowerMsg.includes('you ') && (lowerMsg.includes('never') || lowerMsg.includes('always'))) concerns.push('blame-focused');
    if (lowerMsg.includes('but')) concerns.push('negating statement');
    if (concerns.length === 0) concerns.push('none detected');
    
    const intensity = (lowerMsg.includes('!') || lowerMsg.includes('???')) ? 'high' : 
                     (emotion === 'frustrated' || emotion === 'angry' || emotion === 'stressed') ? 'medium' : 'low';
    
    let reframe = message;
    if (lowerMsg.includes('never')) {
        reframe = message.replace(/you never/gi, "I feel unheard when") + ". Can we talk about this?";
    } else if (lowerMsg.includes('always')) {
        reframe = message.replace(/you always/gi, "I've noticed that sometimes") + ". I'd like to discuss this with you.";
    } else if (lowerMsg.includes('work') || lowerMsg.includes('job')) {
        reframe = "I'm feeling challenged by my current workload. I want to find strategies to manage it better while maintaining my wellbeing.";
    } else {
        reframe = "I'm experiencing [emotion] about [situation]. I'd like to work on [specific goal] to feel better.";
    }
    
    return {
        emotion,
        tone,
        intensity,
        concerns,
        reframe,
        relationship_impact: emotion === 'frustrated' || emotion === 'angry' ? 'negative' : 
                            emotion === 'happy' ? 'positive' : 'neutral',
        communication_quality: concerns.length <= 1 ? 8 : concerns.length === 2 ? 5 : 3
    };
}

function detectVoiceMood(text) {
    const lowerText = text.toLowerCase();
    
    const hasAllCaps = text !== text.toLowerCase() && text !== text.toUpperCase();
    const hasMultiplePunctuation = /[!?]{2,}/.test(text);
    const hasUgh = /\b(ugh|argh|gah|oof)\b/i.test(text);
    
    if (lowerText.includes('stressed') || lowerText.includes('overwhelm') || lowerText.includes('pressure') || lowerText.includes('too much')) {
        return {
            mood: 'Overwhelmed',
            emoji: '😰',
            energy: hasMultiplePunctuation || hasUgh ? 'High' : 'Medium-High',
            style: hasAllCaps ? 'Urgent/Intense' : 'Rapid/Rushed',
            interpretation: "Your words show you're feeling overwhelmed and under significant pressure. The way you expressed this suggests you're reaching your limit.",
            suggestion: "Take a pause right now. Breathe deeply for 30 seconds. Then write down the top 3 things stressing you - often seeing them on paper makes them feel more manageable."
        };
    } else if (lowerText.includes('sad') || lowerText.includes('down') || lowerText.includes('depressed') || lowerText.includes('lonely')) {
        return {
            mood: 'Melancholic',
            emoji: '😔',
            energy: 'Low',
            style: 'Subdued/Reflective',
            interpretation: "Your language indicates sadness and low energy. The words you chose suggest you're going through a difficult emotional time.",
            suggestion: "It's okay to not be okay. Reach out to someone you trust - even a simple 'I'm having a hard day' text can help. Consider doing one small thing that usually comforts you."
        };
    } else if (lowerText.includes('angry') || lowerText.includes('frustrated') || lowerText.includes('annoyed') || lowerText.includes('pissed') || hasUgh) {
        return {
            mood: 'Frustrated',
            emoji: '😤',
            energy: hasMultiplePunctuation ? 'Very High' : 'High',
            style: hasAllCaps ? 'Sharp/Intense' : 'Direct/Blunt',
            interpretation: "Your words convey strong frustration" + (hasMultiplePunctuation ? " and anger" : "") + ". The way you expressed this shows the situation is really getting to you.",
            suggestion: "Before acting on these feelings, try this: go for a 5-minute walk, or write everything you're feeling in a note you won't send. Physical release or private venting can prevent regrettable reactions."
        };
    } else if (lowerText.includes('happy') || lowerText.includes('excited') || lowerText.includes('great') || lowerText.includes('amazing')) {
        return {
            mood: 'Positive',
            emoji: '😊',
            energy: hasMultiplePunctuation ? 'Very High' : 'High',
            style: 'Upbeat/Energetic',
            interpretation: "Your words radiate positive energy! The way you expressed yourself shows genuine enthusiasm and good spirits.",
            suggestion: "This is wonderful! Savor this feeling. Consider journaling what contributed to this mood so you can identify and recreate these positive circumstances."
        };
    } else if (lowerText.includes('worried') || lowerText.includes('anxious') || lowerText.includes('nervous') || lowerText.includes('scared')) {
        return {
            mood: 'Anxious',
            emoji: '😰',
            energy: 'Medium',
            style: 'Hesitant/Uncertain',
            interpretation: "Your language reveals anxiety about the future or a situation. The words you chose show your mind is racing with 'what-ifs'.",
            suggestion: "Try the 5-4-3-2-1 grounding technique: Name 5 things you see, 4 you can touch, 3 you hear, 2 you smell, 1 you taste. This brings you back to the present moment."
        };
    } else if (lowerText.includes('tired') || lowerText.includes('exhausted') || lowerText.includes('drained') || lowerText.includes('burnt out')) {
        return {
            mood: 'Exhausted',
            emoji: '😴',
            energy: 'Very Low',
            style: 'Weary/Depleted',
            interpretation: "Your words indicate deep exhaustion - physical, mental, or both. The way you expressed this suggests you're running on empty.",
            suggestion: "You need rest. If possible, cancel non-essential plans today. Take a nap, go to bed early, or simply sit and do nothing for 20 minutes. Rest isn't lazy - it's necessary."
        };
    } else {
        return {
            mood: 'Thoughtful',
            emoji: '🤔',
            energy: 'Medium',
            style: 'Calm/Measured',
            interpretation: "Your words are reflective and measured. The way you expressed yourself suggests you're processing your thoughts carefully.",
            suggestion: "You seem to be in a good headspace for self-reflection. This is a great time to journal or think deeply about what you want or need."
        };
    }
}

function getMoodColor(mood) {
    const colors = {
        'Overwhelmed': 'rgba(255,152,0,0.3)',
        'Melancholic': 'rgba(33,150,243,0.3)',
        'Frustrated': 'rgba(244,67,54,0.3)',
        'Positive': 'rgba(76,175,80,0.3)',
        'Anxious': 'rgba(156,39,176,0.3)',
        'Exhausted': 'rgba(121,85,72,0.3)',
        'Thoughtful': 'rgba(158,158,158,0.3)'
    };
    return colors[mood] || colors.Thoughtful;
}

function showEmpathyFeedback(index) {
    const feedback = document.getElementById(`feedback-${index}`);
    if (feedback.style.display === 'none') {
        document.querySelectorAll('[id^="feedback-"]').forEach(f => f.style.display = 'none');
        feedback.style.display = 'block';
    } else {
        feedback.style.display = 'none';
    }
}

/******************************************************************************
 * SECTION 17: JAVASCRIPT - MULTIPLE SUGGESTIONS GENERATOR
 ******************************************************************************/

function generateMultipleSuggestions(message, analysis) {
    const contexts = detectProblemContext(message);
    let suggestionsHTML = '<div class="analysis-section"><strong>💡 Personalized Suggestions:</strong></div>';
    
    contexts.forEach(context => {
        const suggestion = getSuggestionByContext(context, message, analysis);
        suggestionsHTML += `
            <div class="suggestion-card ${context}">
                <h5 style="color: #ffd700; margin-bottom: 0.8rem;">
                    ${suggestion.icon} ${suggestion.title}
                </h5>
                <p style="line-height: 1.6; margin-bottom: 0.8rem;">${suggestion.description}</p>
                <div style="background: rgba(255,255,255,0.05); padding: 0.8rem; border-radius: 8px;">
                    <strong>Action Steps:</strong>
                    <ul style="margin: 0.5rem 0 0 1.5rem; line-height: 1.8;">
                        ${suggestion.actions.map(action => `<li>${action}</li>`).join('')}
                    </ul>
                </div>
            </div>
        `;
    });
    
    return suggestionsHTML;
}

function getContextColor(context) {
    const colors = {
        'personal': 'rgba(76,175,80,0.3)',
        'work': 'rgba(33,150,243,0.3)',
        'relationship': 'rgba(233,30,99,0.3)',
        'health': 'rgba(255,152,0,0.3)'
    };
    return colors[context] || 'rgba(158,158,158,0.3)';
}

function getContextIcon(context) {
    const icons = {
        'personal': '🌱',
        'work': '💼',
        'relationship': '💝',
        'health': '🏥'
    };
    return icons[context] || '✨';
}

function getEmotionColor(emotion) {
    const colors = {
        'frustrated': 'rgba(255,152,0,0.3)',
        'sad': 'rgba(33,150,243,0.3)',
        'happy': 'rgba(76,175,80,0.3)',
        'anxious': 'rgba(156,39,176,0.3)',
        'angry': 'rgba(244,67,54,0.3)',
        'stressed': 'rgba(255,193,7,0.3)',
        'neutral': 'rgba(158,158,158,0.3)'
    };
    return colors[emotion] || colors.neutral;
}

    }
}
